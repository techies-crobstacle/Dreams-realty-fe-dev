    7:

   10:     propertyType = "sale",

   38:

   90:     }

  126:             <div className="form-group">
  127:               <label
  128:                 className={`radio-css ${
  129:                   propertyType === "rent" ? "active" : ""
  130:                 }`}
  131:                 htmlFor="radio2"
  132:               >
  133:                 <div>
  134:                   <h4>Rent/Lease</h4>
  135:                   {/* <p>

  146:                     setPropertyType(propertyType === "sale" ? "rent" : "sale")

  160:                 type="text"
  161:                 name="full_name"

  165:                 onChange={(e) => {
  166:                   // Only allow letters and spaces

  175:                   "Full Name",
